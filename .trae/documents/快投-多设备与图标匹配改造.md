```python
def _arp_hosts():
    """读 Windows ARP 邻居表，返回已知的局域网主机 IP（只按 IP/MAC 正则提取，不依赖系统语言）。"""
    out = ""
    try:
        r = subprocess.run(["arp", "-a"], capture_output=True, timeout=8,
                           encoding="utf-8", errors="replace",
                           startupinfo=get_startupinfo(), creationflags=0x08000000)
        out = (r.stdout or "") + (r.stderr or "")
    except Exception:
        return []
    ips = []
    for m in re.finditer(r'(\d{1,3}(?:\.\d{1,3}){3})\s+([0-9a-fA-F]{2}(?:-[0-9a-fA-F]{2}){5})', out):
        ip = m.group(1)
        if ip.endswith(".255") or ip.startswith("224.") or ip.startswith("239."):
            continue
        if ip not in ips:
            ips.append(ip)
    return ips

def _local_prefix():
    """本机局域网 /24 网段前缀，如 '192.168.1'；取不到返回 None。"""
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        s.connect(("223.5.5.5", 80))
        ip = s.getsockname()[0]
    except Exception:
        ip = None
    finally:
        s.close()
    if not ip or ip.startswith("127."):
        return None
    return ip.rsplit(".", 1)[0]

def _probe_5555(ips):
    """并发探测一批 IP 的 5555 端口，返回开放的 IP 列表（保持输入顺序）。"""
    if not ips:
        return []
    hits = []
    try:
        with concurrent.futures.ThreadPoolExecutor(max_workers=_PROBE_WORKERS) as ex:
            for ip, ok in zip(ips, ex.map(lambda a: _probe_port(a, CLASSIC_ADB_PORT), ips, chunksize=8)):
                if ok:
                    hits.append(ip)
    except Exception:
        pass
    return hits
```

```python
_icon_index = {}            # basename(lower) -> 绝对路径
_icon_index_lock = threading.Lock()

def _icon_index_scan():
    idx = {}
    for d in ICON_SEARCH_DIRS:
        try:
            for name in os.listdir(d):
                stem, ext = os.path.splitext(name)
                if ext.lower() not in ICON_EXTS:
                    continue
                p = os.path.join(d, name)
                try:
                    if os.path.getsize(p) <= 0:
                        continue
                except OSError:
                    continue
                idx.setdefault(stem.lower(), p)     # 可写目录优先
        except OSError:
            continue
    return idx

def _icon_index_add(path):
    stem, ext = os.path.splitext(os.path.basename(path))
    if ext.lower() in ICON_EXTS:
        with _icon_index_lock:
            _icon_index.setdefault(stem.lower(), path)

def _norm_key(s):
    return re.sub(r'[^a-z0-9]', '', s.lower())

def find_cached_icon(pkg):
    """本地素材库匹配：精确 → 归一化 → 互为后缀 → 尾段重合。全部基于已建立的索引，O(1)/O(n) 且不碰网络。"""
    key = (pkg or "").strip().lower()
    if not key:
        return None
    with _icon_index_lock:
        idx = _icon_index
    hit = idx.get(key)
    if hit:
        return hit
    nk = _norm_key(key)
    if nk:
        for k, p in idx.items():
            if _norm_key(k) == nk:
                return p
    segs = key.split(".")
    best, best_len = None, -1
    for k, p in idx.items():
        ks = k.split(".")
        # 互为后缀：短的必须是长的末尾若干段，且短的至少 2 段
        if len(ks) >= 2 and (key.endswith("." + k) or k.endswith("." + key)):
            if len(ks) > best_len:
                best, best_len = p, len(ks)
            continue
        # 尾段重合：最后两段相同，且公共后缀 >= 2 段，取公共后缀最长者
        if len(segs) >= 2 and len(ks) >= 2 and segs[-1] == ks[-1] and segs[-2] == ks[-2]:
            common = 0
            while common < min(len(segs), len(ks)) and segs[-1 - common] == ks[-1 - common]:
                common += 1
            if common > best_len:
                best, best_len = p, common
    return best
```

### D. `launcher_server.py` — dpi 同步电脑 dpi

```python
def _pc_dpi():
    """取电脑主显示器的逻辑 DPI（LOGPIXELSY，常见 96/120/144），失败回退 96。"""
    try:
        hdc = ctypes.windll.user32.GetDC(0)
        try:
            dpi = ctypes.windll.gdi32.GetDeviceCaps(hdc, 90)   # LOGPIXELSY
        finally:
            ctypes.windll.user32.ReleaseDC(0, hdc)
        return int(dpi) if dpi and 60 <= dpi <= 480 else 96
    except Exception:
        return 96
```

- `build_scrcpy_cmd()` 中：`dpi = max(72, int(round(_pc_dpi() * scale)))`，`scale` 默认 `1.0`（保持配置项与设置里的「UI 放大倍数」）。
- 其余参数（`--flex-display` / 码率 / 帧率 / 音频模式）不变。

## Assumptions & Decisions

1. 设备发现的「5555 端口」前提：手机需已 `adb tcpip 5555`（经典无线调试）。已连接的设备（可能是随机端口）仍照常列出，不会被 5555 过滤掉。
2. 扫描阶段**不做任何 adb connect**，从根本上避免「扫描把第二台设备连上导致 Multiple ADB devices」。
3. `icons/` 素材库的键就是包名（已核实 2277 个文件均为包名形态的 ASCII 文件名），因此「提高匹配率」= 在包名上做容错匹配，不做「应用名 → 包名」猜测。
4. 在线取图标（应用宝 / 小米 / iTunes）**保留**，但只作为后台队列，且 `/api/icon` 不再等待它；用户要求的「取消从手机获取图标」仅指删除 adb/APK 那条链路。
5. 旧 `apps_cache.json`（单设备格式）与旧 `config.json` 的 list 型 `quick_launch` 不做完整兼容：`apps_cache` 丢弃重扫；`quick_launch` 的 list 迁移为 `{"*": [...]}` 作为兜底默认。
6. 投屏参数（分辨率/码率/帧率/音频/缩放）全局共用一份，不按设备分开存。
7. 不新增「关于」等未提及的设置项与页面。
